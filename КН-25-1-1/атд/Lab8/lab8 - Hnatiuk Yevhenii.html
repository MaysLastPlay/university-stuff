import networkx as nx
import matplotlib.pyplot as plt
G = nx.DiGraph()
edges = [
    ('V0', 'V1', 5), ('V0', 'V5', 2), 
    ('V1', 'V2', 4), 
    ('V2', 'V3', 9), 
    ('V3', 'V4', 7), ('V3', 'V5', 3), 
    ('V4', 'V0', 1), 
    ('V5', 'V4', 8), ('V5', 'V2', 1)
]
G.add_weighted_edges_from(edges)

pos = nx.spring_layout(G, seed=42)

plt.figure(figsize=(8, 6))
nx.draw_networkx_nodes(G, pos, node_size=700, node_color='lightblue')
nx.draw_networkx_labels(G, pos, font_size=15, font_family='sans-serif')
nx.draw_networkx_edges(G, pos, edgelist=G.edges(), arrows=True, arrowsize=20)
edge_labels = nx.get_edge_attributes(G, 'weight')
nx.draw_networkx_edge_labels(G, pos, edge_labels=edge_labels, font_size=12)
plt.axis('off')
plt.show()

adj_dict = {node: list(G.successors(node)) for node in G.nodes()}

def dfs_paths(graph_dict, start, goal, path=None):
    if path is None:
        path = [start]
    if start == goal:
        yield path
    for next_node in set(graph_dict[start]) - set(path):
        yield from dfs_paths(graph_dict, next_node, goal, path + [next_node])

def bfs_paths(graph_dict, start, goal):
    queue = [(start, [start])]
    while queue:
        vertex, path = queue.pop(0)
        for next_node in set(graph_dict[vertex]) - set(path):
            if next_node == goal:
                yield path + [next_node]
            else:
                queue.append((next_node, path + [next_node]))

start_v, goal_v = 'V0', 'V4'

print(f"\n--- Всі шляхи від {start_v} до {goal_v} ---")
print("DFS шляхи:", list(dfs_paths(adj_dict, start_v, goal_v)))
print("BFS шляхи:", list(bfs_paths(adj_dict, start_v, goal_v)))

Контрольні запитання:¶

  1. Що таке граф? Граф — це абстрактна структура даних, що складається з множини вершин (вузлів) та множини ребер, які їх з'єднують. Застосовується для моделювання комп'ютерних мереж, маршрутів навігації (GPS), соціальних мереж.
  2. Види графів: Основні види — орієнтовані (ребра мають напрямок від однієї вершини до іншої) та неорієнтовані (ребра є двосторонніми). Також бувають зважені (ребра мають "вартість" або відстань) і незважені.
  3. Представлення в пам'яті: Найчастіше графи зберігають у вигляді матриці суміжності (двовимірний масив, зручно для щільних графів, $O(V^2)$ пам'яті) або списку суміжності (словник, де ключі — вершини, а значення — списки сусідів, ідеально для розріджених графів, $O(V+E)$ пам'яті).
  4. BFS (у ширину): Досліджує граф по рівнях — спочатку всіх найближчих сусідів початкової вершини, потім сусідів сусідів. Працює на основі Черги (Queue). Застосовується для пошуку найкоротшого шляху в незважених графах.
  5. DFS (у глибину): Йде по одній гілці графа "вглиб" до самого кінця, а потім повертається (backtracking). Працює на основі Стеку або рекурсії. Використовується для знаходження виходу з лабіринту, топологічного сортування, пошуку циклів.
  6. Алгоритм Дейкстри: Знаходить найкоротший шлях від початкової вершини до всіх інших. Головна умова успішної роботи — у графі не повинно бути ребер з від'ємною вагою. Якщо такі ребра є, слід використовувати алгоритм Беллмана-Форда.
